# CPIA PoC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 작업일보 파일 드롭 → Collector→Parser→Mapper→Atomizer→Reporter‖Forecaster→Optimizer → Streamlit 대시보드까지 전체 7-Agent 파이프라인을 라이브 시연 가능한 수준으로 구현한다.

**Architecture:** LangGraph 기반 7-Agent 순차+병렬 파이프라인. Mapper 완료 후 Reporter와 Forecaster가 병렬 실행, Forecaster→H-i-L 2→Optimizer 순차 실행. ChromaDB(로컬)가 Atomizer(적재)·Forecaster·Optimizer(조회) 사이의 공유 Vector Store 역할.

**Tech Stack:** Python 3.11+, LangGraph, anthropic SDK (Sonnet 4.6 / Opus 4.7), chromadb, openai (embedding), streamlit, openpyxl, pytest

```python
# scripts/extract_wbs_data.py
import openpyxl, json, sys
from pathlib import Path

WBS_FILE = Path("docs/sample/WBS공정표_M15X IBL PH-3_CPRS_2025.09.25.xlsx")

def extract_wbs_list():
    wb = openpyxl.load_workbook(WBS_FILE, data_only=True)
    ws = wb["Activity List(입력)"]
    
    trades = set()
    activities = []
    
    for row in ws.iter_rows(min_row=6, values_only=True):  # 헤더 5행 스킵
        if not row[1]:  # Activity ID 없으면 종료
            continue
        activity_id = str(row[1]).strip()
        work_area   = str(row[4]).strip() if row[4] else ""
        trade       = str(row[7]).strip() if row[7] else ""   # 열 번호는 실행 후 조정
        scope       = str(row[8]).strip() if row[8] else ""
        
        if activity_id and trade:
            trades.add(trade)
            activities.append({
                "activity_id": activity_id,
                "work_area": work_area,
                "trade": trade,
                "scope": scope
            })
    
    return list(trades), activities


def extract_remaining_qty():
    wb = openpyxl.load_workbook(WBS_FILE, data_only=True)
    ws = wb["잔여물량"]
    
    remaining = []
    for row in ws.iter_rows(min_row=6, values_only=True):
        if not row[1]:
            continue
        remaining.append({
            "work_area": str(row[0]).strip() if row[0] else "",
            "trade":     str(row[1]).strip() if row[1] else "",
            "code":      str(row[2]).strip() if row[2] else "",
            "plan_start": str(row[3]).strip() if row[3] else "",
            "plan_end":   str(row[4]).strip() if row[4] else "",
        })
    return remaining


if __name__ == "__main__":
    trades, activities = extract_wbs_list()
    remaining = extract_remaining_qty()
    
    Path("data/wbs_list.json").write_text(
        json.dumps({"trades": sorted(trades), "activities": activities[:200]},  # 대표 200개
                   ensure_ascii=False, indent=2)
    )
    Path("data/remaining_qty.json").write_text(
        json.dumps(remaining, ensure_ascii=False, indent=2)
    )
    print(f"공종 {len(trades)}개, Activity {len(activities)}개, 잔여물량 {len(remaining)}개 추출 완료")
```

# src/pipeline/state.py
from typing import TypedDict, Annotated
import operator


class ActivityRecord(TypedDict):
    wbs_code: str           # Mapper가 채움 (초기: "")
    trade: str              # 공종명 (거푸집, 철근, 콘크리트 등)
    workers: int            # 투입 인원
    quantity: float         # 당일 물량
    unit: str               # 단위 (m², ton, m³)
    work_content: str       # 작업 내용 전문 (임베딩 핵심)
    conditions: list[str]   # ["우천", "야간연장없음"] 등
    requires_review: bool   # Mapper 확신도 낮음 플래그


class DailyReport(TypedDict):
    date: str
    bp_code: str
    bp_name: str
    activities: list[ActivityRecord]


class ForecastResult(TypedDict):
    trade: str
    today_productivity: float
    unit: str
    assessment: str         # "정상" | "주의" | "이상"
    reasoning: str
    similar_cases: list[str]
    expected_completion: str
    delay_days: int
    risk_level: str


class PipelineState(TypedDict):
    # Collector
    raw_files: list[str]           # 수집된 파일 경로 목록
    raw_texts: dict[str, str]      # {파일명: 추출 텍스트}
    
    # Parser
    parsed_reports: list[DailyReport]
    
    # Mapper
    mapped_reports: list[DailyReport]
    mapping_corrections: dict      # H-i-L 1 수정 내역
    hil1_approved: bool
    
    # Atomizer
    atomized_count: int
    
    # Reporter
    daily_report_path: str
    
    # Forecaster
    forecast_results: list[ForecastResult]
    risk_trades: list[str]         # 이상 판정 공종 목록
    
    # H-i-L 2
    selected_trades: list[str]     # PM이 선택한 조언 대상 공종
    hil2_approved: bool
    
    # Optimizer
    optimizer_advice: dict         # {공종: 조언 텍스트}
    
    # 공통
    date: str
    error: str | None
```

    },
}

def generate_record(date_str, contractor, trade, prev_conditions, prev_overtime):
    conditions = pick_conditions(prev_conditions)
    base = BASE_PRODUCTIVITY[trade]
    productivity = calc_productivity(base, conditions, prev_overtime)
    headcount = random.randint(10, 20)
    output = round(productivity * headcount, 1)
    floor = random.choice(["1", "2", "3", "B1", "B2"])
    cond_key = conditions[0] if conditions else "정상"
    content = WORK_CONTENT_TEMPLATES[trade].get(cond_key, WORK_CONTENT_TEMPLATES[trade]["정상"])
    
    return {
        "date":          date_str,
        "contractor":    contractor,
        "trade":         trade,
        "headcount":     headcount,
        "output":        output,
        "unit":          UNITS[trade],
        "productivity":  productivity,
        "conditions":    conditions,
        "work_content":  content.format(floor=floor),
    }

if __name__ == "__main__":
    out_dir = Path("data/synthetic_history")
    out_dir.mkdir(exist_ok=True)
    
    count = 0
    for contractor in CONTRACTORS:
        for trade in TARGET_TRADES:
            prev_conditions = ["정상"]
            prev_overtime   = False
            
            for day in range(DAYS):
                current_date = START_DATE + timedelta(days=day)
                if current_date.weekday() >= 6:  # 일요일 제외
                    continue
                
                date_str = current_date.strftime("%Y-%m-%d")
                record = generate_record(date_str, contractor, trade, prev_conditions, prev_overtime)
                
                filename = f"{date_str}_{contractor}_{trade}.json"
                (out_dir / filename).write_text(json.dumps(record, ensure_ascii=False, indent=2))
                
                prev_conditions = record["conditions"]
                prev_overtime   = "야간연장_50초과" in record["conditions"]
                count += 1
    
    print(f"Synthetic history {count}개 레코드 생성 완료 → data/synthetic_history/")

# scripts/init_vector_db.py
import json, chromadb
from pathlib import Path
from openai import OpenAI
from dotenv import load_dotenv

load_dotenv()

CHROMA_PATH     = "data/vector_db"
COLLECTION_NAME = "cpia_work_history"
EMBEDDING_MODEL = "text-embedding-3-small"


def embed_texts(texts: list[str], client: OpenAI) -> list[list[float]]:
    response = client.embeddings.create(model=EMBEDDING_MODEL, input=texts)
    return [item.embedding for item in response.data]


def load_synthetic_records() -> list[dict]:
    records = []
    for f in sorted(Path("data/synthetic_history").glob("*.json")):
        records.append(json.loads(f.read_text()))
    return records


if __name__ == "__main__":
    openai_client  = OpenAI()
    chroma_client  = chromadb.PersistentClient(path=CHROMA_PATH)
    
    # 기존 컬렉션 삭제 후 재생성 (초기화)
    try:
        chroma_client.delete_collection(COLLECTION_NAME)
    except Exception:
        pass
    collection = chroma_client.create_collection(
        COLLECTION_NAME,
        metadata={"hnsw:space": "cosine"}
    )
    
    records = load_synthetic_records()
    print(f"{len(records)}개 레코드 임베딩 중...")
    
    # 배치 처리 (OpenAI 임베딩 API 배치 한계 고려)
    BATCH = 50
    for start in range(0, len(records), BATCH):
        batch = records[start:start + BATCH]
        texts = [r["work_content"] for r in batch]
        embeddings = embed_texts(texts, openai_client)
        
        collection.add(
            ids          = [f"{r['date']}_{r['contractor']}_{r['trade']}" for r in batch],
            embeddings   = embeddings,

                "trade":        r["trade"],
                "headcount":    r["headcount"],
                "output":       r["output"],
                "unit":         r["unit"],
                "productivity": r["productivity"],
                "conditions":   ",".join(r["conditions"]),
            } for r in batch],
        )
        print(f"  {start + len(batch)}/{len(records)} 적재 완료")
    
    print(f"ChromaDB 초기화 완료. 총 {collection.count()}개 레코드.")

# src/agents/collector.py
import openpyxl
from pathlib import Path


def collect_files(folder: str) -> list[str]:
    p = Path(folder)
    return [str(f) for f in p.glob("**/*") if f.suffix in (".xlsx", ".xlsm", ".txt")]


def extract_text_from_excel(file_path: str) -> dict:
    wb = openpyxl.load_workbook(file_path, data_only=True)
    sheets = {}
    for name in wb.sheetnames:
        ws = wb[name]
        lines = []
        for row in ws.iter_rows(values_only=True):
            cells = [str(c).strip() for c in row if c is not None and str(c).strip()]
            if cells:
                lines.append(" | ".join(cells))
        sheets[name] = "\n".join(lines)
    return {"file": file_path, "sheets": sheets}
```

WBS_TRADES_HINT = """대상 공종 목록 (거푸집, 철근, 콘크리트 — 다른 공종은 무시)"""

PARSER_PROMPT = """당신은 건설 현장 공정 데이터 추출 전문가입니다.
아래 작업일보 Excel 추출 텍스트에서 다음 정보를 JSON으로 추출하세요.

대상 공종: {target_trades}
작업 날짜: {date}

추출 규칙:
1. "4.물량" 시트에서 공종별 금일 물량(quantity)과 투입 인원(workers)을 찾으세요.
2. "1.출력인원 현황" 시트의 "금일 작업사항" 텍스트를 work_content에 원문 그대로 보존하세요.
   (이 텍스트는 AI가 생산성 원인 분석에 활용하므로 절대 요약하지 말 것)
3. 날씨·작업 여건 정보는 conditions 배열에 ["우천", "장비대기", "병행작업", "야간연장", "정상"] 중 선택.
4. BP코드(PH_X-NNN)와 업체명을 식별하세요.

출력 형식 (JSON만, 설명 없이):
{{
  "date": "{date}",
  "bp_code": "PH_X-XXX",
  "bp_name": "업체명",
  "activities": [
    {{
      "wbs_code": "",
      "trade": "공종명",
      "workers": 14,
      "quantity": 320.0,
      "unit": "m²",
      "work_content": "원문 작업내용 전문",
      "conditions": ["우천"],
      "requires_review": false
    }}
  ]
}}

작업일보 원문:
---
{raw_text}
---"""


def parse_daily_report(extracted: dict, date: str) -> DailyReport:
    from src.config import TARGET_TRADES
    
    sheets_text = "\n\n".join(
        f"[{name}]\n{text}" for name, text in extracted["sheets"].items()
        if name in ("4.물량", "1.출력인원 현황", "2.직종별 인원현황")
    )
    
    client = anthropic.Anthropic()
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=4096,
        messages=[{
            "role": "user",
            "content": PARSER_PROMPT.format(
                target_trades=", ".join(TARGET_TRADES),
                date=date,
                raw_text=sheets_text[:8000]  # 토큰 절약
            )
        }]
    )
    
    raw = response.content[0].text.strip()
    # JSON 블록 추출 (LLM이 ```json ... ``` 으로 감쌀 수 있음)
    if "```" in raw:
        raw = raw.split("```")[1].replace("json", "", 1).strip()
    
    return json.loads(raw)
```

def atomize_and_store(report: DailyReport) -> int:

    collection    = chroma_client.get_collection(COLLECTION_NAME)

    for activity in report["activities"]:
        if not activity["work_content"]:
            continue
        
        record_id = f"{report['date']}_{report['bp_code']}_{activity['trade']}"
        productivity = round(activity["quantity"] / activity["workers"], 2) if activity["workers"] > 0 else 0

            model=EMBEDDING_MODEL,
            input=activity["work_content"]
        )
        embedding = embed_response.data[0].embedding
        
        collection.upsert(
            ids        = [record_id],
            embeddings = [embedding],
            documents  = [activity["work_content"]],
            metadatas  = [{
                "date":         report["date"],
                "contractor":   report["bp_name"],
                "trade":        activity["trade"],
                "headcount":    activity["workers"],
                "output":       activity["quantity"],
                "unit":         activity["unit"],
                "productivity": productivity,
                "conditions":   ",".join(activity.get("conditions", [])),
            }]
        )
        stored += 1

        for ri, r in enumerate(state["mapped_reports"])
        for ai, a in enumerate(r["activities"])
        if a.get("requires_review")
    ]
    
    if not flagged:
        return {**state, "hil1_approved": True}
    
    print(f"\n[H-i-L 1] 매핑 검토 필요 항목 {len(flagged)}건:")
    for ri, ai, r, a in flagged:
        print(f"  [{r['bp_name']}] {a['trade']}: 현재 매핑={a['wbs_code']} (작업내용: {a['work_content'][:60]})")
    
    user_input = interrupt({"flagged": [
        {"bp_name": r["bp_name"], "trade": a["trade"], "current_wbs": a["wbs_code"]}
        for _, _, r, a in flagged
    ]})
    
    return {**state, "mapping_corrections": user_input, "hil1_approved": True}


def node_atomizer(state: PipelineState) -> PipelineState:
    count = sum(atomize_and_store(r) for r in state["mapped_reports"])
    return {**state, "atomized_count": count}


def node_reporter(state: PipelineState) -> PipelineState:
    path = generate_report(state["mapped_reports"], state["date"])
    return {**state, "daily_report_path": path}


def node_forecaster(state: PipelineState) -> PipelineState:
    # Atomizer 완료 후 실행 (병렬 아님 — Atomizer가 오늘 데이터 적재 후 조회해야 함)
    all_forecasts = []
    for r in state["mapped_reports"]:
        all_forecasts.extend(diagnose_productivity(r))
    
    risk_trades = [f["trade"] for f in all_forecasts if f["assessment"] in ("주의", "이상")]
    return {**state, "forecast_results": all_forecasts, "risk_trades": risk_trades}


def node_hil2(state: PipelineState) -> PipelineState:
    from langgraph.types import interrupt
    
    if not state.get("risk_trades"):
        return {**state, "selected_trades": [], "hil2_approved": True}
    
    print(f"\n[H-i-L 2] 생산성 이상 공종: {state['risk_trades']}")
    print("Optimizer 조언이 필요한 공종을 선택하세요.")
    
    selected = interrupt({"risk_trades": state["risk_trades"]})
    return {**state, "selected_trades": selected, "hil2_approved": True}


def node_optimizer(state: PipelineState) -> PipelineState:
    if not state.get("selected_trades"):
        return {**state, "optimizer_advice": {}}
    
    advice = generate_advice(state["selected_trades"])
    return {**state, "optimizer_advice": advice}


def build_pipeline():
    graph = StateGraph(PipelineState)
    
    graph.add_node("collector",  node_collector)
    graph.add_node("parser",     node_parser)
    graph.add_node("mapper",     node_mapper)
    graph.add_node("hil1",       node_hil1)
    graph.add_node("atomizer",   node_atomizer)
    graph.add_node("reporter",   node_reporter)
    graph.add_node("forecaster", node_forecaster)
    graph.add_node("hil2",       node_hil2)
    graph.add_node("optimizer",  node_optimizer)
    
    graph.set_entry_point("collector")
    graph.add_edge("collector",  "parser")
    graph.add_edge("parser",     "mapper")
    graph.add_edge("mapper",     "hil1")
    graph.add_edge("hil1",       "atomizer")
    graph.add_edge("atomizer",   "reporter")
    graph.add_edge("reporter",   "forecaster")
    graph.add_edge("forecaster", "hil2")
    graph.add_edge("hil2",       "optimizer")
    graph.add_edge("optimizer",  END)
    
    return graph.compile(checkpointer=MemorySaver(), interrupt_before=["hil1", "hil2"])
```

# main.py
import sys
from datetime import date
from src.pipeline.graph import build_pipeline

def run_pipeline(input_date: str = None):
    target_date = input_date or date.today().strftime("%Y%m%d")
    
    pipeline = build_pipeline()
    config   = {"configurable": {"thread_id": target_date}}
    
    initial_state = {
        "date": target_date,
        "raw_files": [], "raw_texts": {},
        "parsed_reports": [], "mapped_reports": [],
        "mapping_corrections": {}, "hil1_approved": False,
        "atomized_count": 0, "daily_report_path": "",
        "forecast_results": [], "risk_trades": [],
        "selected_trades": [], "hil2_approved": False,
        "optimizer_advice": {}, "error": None,
    }
    
    print(f"=== CPIA 파이프라인 시작: {target_date} ===")
    
    for event in pipeline.stream(initial_state, config):
        for node_name, result in event.items():
            if node_name == "__interrupt__":
                interrupt_data = result[0].value
                print(f"\n[중단점] {interrupt_data}")
                user_input = input("입력 (엔터 = 승인): ").strip() or []
                pipeline.invoke({"resume": user_input}, config)
            else:
                print(f"  ✓ {node_name} 완료")
    
    print(f"\n=== 파이프라인 완료 ===")


if __name__ == "__main__":
    date_arg = sys.argv[1] if len(sys.argv) > 1 else None
    run_pipeline(date_arg)

# src/dashboard/app.py
import json, csv, streamlit as st
from pathlib import Path

st.set_page_config(page_title="CPIA 현장 현황", layout="wide")

def load_json(path: str) -> dict | list | None:
    p = Path(path)
    return json.loads(p.read_text()) if p.exists() else None

# ── Layer 1: 6개 현장 전체 현황 ─────────────────────────────────────
st.title("CPIA 현장 현황 대시보드")
st.caption(f"마지막 갱신: {Path('data/forecast.json').stat().st_mtime if Path('data/forecast.json').exists() else 'N/A'}")

mock_sites = load_json("data/mock_sites.json") or []
forecast   = load_json("data/forecast.json") or []

# Demo Site는 실데이터로 교체
demo_site = {
    "site_id": "DEMO", "name": "Demo Site (실데이터)",
    "plan_progress": 38.5,
    "actual_progress": 34.2,
    "risk_trades": [f["trade"] for f in forecast if f.get("assessment") in ("주의", "이상")],
    "last_report": "오늘 자동 생성" if Path("data/forecast.json").exists() else "미생성",
}
all_sites = [demo_site] + [s for s in mock_sites if s["site_id"] != "DEMO"]

st.subheader("Layer 1 — 전체 현장 현황")
cols = st.columns(3)
for i, site in enumerate(all_sites):
    with cols[i % 3]:
        delta = site["actual_progress"] - site["plan_progress"]
        delta_str = f"{delta:+.1f}%p"
        color = "🔴" if delta < -3 else ("🟡" if delta < 0 else "🟢")
        
        with st.container(border=True):
            st.markdown(f"**{color} {site['name']}**")
            st.metric("공정률", f"{site['actual_progress']:.1f}%", delta_str)
            if site["risk_trades"]:
                st.warning(f"이상 공종: {', '.join(site['risk_trades'])}")
            st.caption(f"보고: {site['last_report']}")
            if st.button("상세 보기", key=site["site_id"]):
                st.session_state["selected_site"] = site["site_id"]

# ── Layer 2: 현장 상세 (Demo Site만 실데이터) ─────────────────────────
selected = st.session_state.get("selected_site")
if selected == "DEMO":
    st.divider()
    st.subheader("Layer 2 — Demo Site 상세")
    
    # 공정 현황
    col1, col2 = st.columns(2)
    with col1:
        st.metric("계획 공정률", "38.5%")
    with col2:
        st.metric("실적 공정률", "34.2%", "-4.3%p")
    
    # Forecaster 진단 결과
    if forecast:
        st.markdown("#### 생산성 진단 결과")
        for fc in forecast:
            icon = "✅" if fc["assessment"] == "정상" else ("⚠️" if fc["assessment"] == "주의" else "🚨")
            with st.expander(f"{icon} {fc['trade']} — {fc['assessment']} ({fc['today_productivity']} {fc['unit']})"):
                st.write(fc.get("reasoning", ""))
                if fc.get("similar_cases"):
                    st.caption("유사 사례: " + " / ".join(fc["similar_cases"]))
                st.caption(f"예상 완료: {fc.get('expected_completion', '-')} (지연 {fc.get('delay_days', 0)}일)")

                st.write(advice)
    
    # 보고서 링크

    if reports:

        for r in reports[-5:]:
            st.download_button(f"📄 {r.name}", r.read_text(), file_name=r.name)
    
    # H-i-L 버튼 (파이프라인 실행 트리거)
    st.divider()
    col1, col2 = st.columns(2)
    with col1:
        if st.button("🚀 파이프라인 실행", type="primary"):
            import subprocess
            subprocess.Popen(["python", "main.py"])
            st.success("파이프라인 시작됨 — 5분 후 새로고침하세요")
    with col2:
        if st.button("🔄 대시보드 갱신"):
            st.rerun()
